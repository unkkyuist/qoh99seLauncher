cmake_minimum_required(VERSION 3.20)
project(QOHLauncher LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(QOH-Launcher WIN32 main.cpp launcher.manifest)
target_compile_definitions(QOH-Launcher PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(QOH-Launcher PRIVATE /utf-8 /W4 /EHsc)
target_link_libraries(QOH-Launcher PRIVATE comctl32 bcrypt shell32)
set_property(TARGET QOH-Launcher PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(native_keys_test native_keys_test.cpp)
target_compile_options(native_keys_test PRIVATE /utf-8 /W4 /EHsc)
enable_testing()
add_test(NAME native_keys COMMAND native_keys_test)
add_executable(filter_plan_test filter_plan_test.cpp)
target_compile_options(filter_plan_test PRIVATE /utf-8 /W4 /EHsc)
add_test(NAME filter_plan COMMAND filter_plan_test)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/assets/app.ico")
    enable_language(RC)
    target_sources(QOH-Launcher PRIVATE launcher.rc)
endif()
