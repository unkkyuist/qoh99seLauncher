cmake_minimum_required(VERSION 3.20)
project(QOHLauncher LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(RUNTIME_DIR "${CMAKE_CURRENT_BINARY_DIR}/runtime")
add_custom_target(prepare_runtime
    COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/prepare-runtime.ps1" -OutputDir "${RUNTIME_DIR}"
    BYPRODUCTS "${RUNTIME_DIR}/runtime_payload.rc" "${RUNTIME_DIR}/runtime_payload.h"
    VERBATIM)
add_executable(QOH-Launcher WIN32 main.cpp portable_runtime.cpp launcher.manifest "${RUNTIME_DIR}/runtime_payload.rc")
add_dependencies(QOH-Launcher prepare_runtime)
target_include_directories(QOH-Launcher PRIVATE "${RUNTIME_DIR}")
target_compile_definitions(QOH-Launcher PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(QOH-Launcher PRIVATE /utf-8 /W4 /EHsc)
target_link_libraries(QOH-Launcher PRIVATE comctl32 bcrypt shell32)
set_property(TARGET QOH-Launcher PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(native_keys_test native_keys_test.cpp)
target_compile_options(native_keys_test PRIVATE /utf-8 /W4 /EHsc)
enable_testing()
add_test(NAME native_keys COMMAND native_keys_test)
add_executable(filter_plan_test filter_plan_test.cpp)
target_compile_options(filter_plan_test PRIVATE /utf-8 /W4 /EHsc)
add_test(NAME filter_plan COMMAND filter_plan_test)
add_executable(display_plan_test display_plan_test.cpp)
target_compile_options(display_plan_test PRIVATE /utf-8 /W4 /EHsc)
add_test(NAME display_plan COMMAND display_plan_test)
add_executable(portable_runtime_test portable_runtime_test.cpp portable_runtime.cpp "${RUNTIME_DIR}/runtime_payload.rc")
add_dependencies(portable_runtime_test prepare_runtime)
target_include_directories(portable_runtime_test PRIVATE "${RUNTIME_DIR}")
target_compile_definitions(portable_runtime_test PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(portable_runtime_test PRIVATE /utf-8 /W4 /EHsc)
add_test(NAME portable_runtime COMMAND portable_runtime_test)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/assets/app.ico")
    enable_language(RC)
    target_sources(QOH-Launcher PRIVATE launcher.rc)
endif()
